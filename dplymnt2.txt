  const routeIds = ["home", "digital-lab", "sound-motion", "consultation", "contact"];
  const navlinks = document.getElementById("navlinks");
  const menuToggle = document.getElementById("menuToggle");

  function renderRoute() {
    const route = (location.hash || "#home").slice(1);
    const current = routeIds.includes(route) ? route : "home";
    document.querySelectorAll(".view").forEach(el => el.classList.add("hidden"));
    const view = document.getElementById("view-" + current);
    if (view) view.classList.remove("hidden");
    document.querySelectorAll("[data-route]").forEach(link => {
      link.classList.toggle("active", link.dataset.route === current);
      if (link.closest("#navlinks")) link.setAttribute("aria-current", link.dataset.route === current ? "page" : "false");
    });
    navlinks.classList.remove("open");
    menuToggle.setAttribute("aria-expanded", "false");
    window.scrollTo({ top: 0, behavior: "instant" });
  }

  document.querySelectorAll("[data-route]").forEach(link => {
    link.addEventListener("click", () => {
      if (location.hash === link.getAttribute("href")) renderRoute();
    });
  });
  window.addEventListener("hashchange", renderRoute);
  menuToggle.addEventListener("click", () => {
    const open = navlinks.classList.toggle("open");
    menuToggle.setAttribute("aria-expanded", String(open));
  });

  document.getElementById("year").textContent = new Date().getFullYear();

  // ===== Digital Lab Presentation =====
  (function() {
    const slides = document.querySelectorAll("#labPresentation .slide");
    const total = slides.length;
    let current = 0;

    // Project screenshot data — easy to edit later
    const projectShots = {
      coviat: [
        "assets/digital-lab/coviat-os-1.jpg",
        "assets/digital-lab/coviat-os-2.jpg"
      ],
      oracle: [
        "assets/digital-lab/oracle-gpt-1.jpg",
        "assets/digital-lab/oracle-gpt-2.jpg"
      ],
      xpro: [
        "assets/digital-lab/xpro-1.jpg",
        "assets/digital-lab/xpro-2.jpg"
      ],
      corveux: [
        "assets/digital-lab/corveux-1.jpg",
        "assets/digital-lab/corveux-2.jpg"
      ]
    };

    const shotIndex = { coviat: 0, oracle: 0, xpro: 0, corveux: 0 };

    // Build main progress dots
    const dotsContainer = document.getElementById("presDots");
    for (let i = 0; i < total; i++) {
      const btn = document.createElement("button");
      btn.className = "pres-dot" + (i === 0 ? " active" : "");
      btn.setAttribute("aria-label", "Go to slide " + (i + 1));
      btn.addEventListener("click", () => goTo(i));
      dotsContainer.appendChild(btn);
    }

    function goTo(index) {
      if (index < 0 || index >= total) return;
      slides[current].classList.remove("active");
      current = index;
      slides[current].classList.add("active");
      document.querySelectorAll(".pres-dot").forEach((d, i) => {
        d.classList.toggle("active", i === current);
      });
      document.getElementById("presPrev").disabled = current === 0;
      document.getElementById("presNext").disabled = current === total - 1;
    }

    document.getElementById("presPrev").addEventListener("click", () => goTo(current - 1));
    document.getElementById("presNext").addEventListener("click", () => goTo(current + 1));

    // Screenshot navigation per project
    function updateShot(project) {
      const stage = document.querySelector(`.screenshot-stage[data-project="${project}"]`);
      if (!stage) return;
      const imgs = projectShots[project];
      const idx = shotIndex[project];
      const img = stage.querySelector("img");
      const placeholder = stage.querySelector(".screenshot-placeholder");

      if (img) {
        img.src = imgs[idx];
        img.alt = project + " screenshot " + (idx + 1);
        img.style.display = "";
        if (placeholder) placeholder.style.display = "none";
        img.onerror = function() {
          this.style.display = "none";
          if (placeholder) placeholder.style.display = "flex";
        };
      }

      // Update dots
      const dots = document.querySelector(`[data-shot-dots="${project}"]`);
      if (dots) {
        dots.innerHTML = "";
        imgs.forEach((_, i) => {
          const d = document.createElement("button");
          d.className = "shot-dot" + (i === idx ? " active" : "");
          d.addEventListener("click", () => {
            shotIndex[project] = i;
            updateShot(project);
          });
          dots.appendChild(d);
        });
      }
    }

    // Initialize screenshot dots and attach prev/next
    Object.keys(projectShots).forEach(project => {
      updateShot(project);
      const stage = document.querySelector(`.screenshot-stage[data-project="${project}"]`);
      if (!stage) return;
      const nav = stage.parentElement.querySelector(".screenshot-nav");
      if (!nav) return;
      nav.querySelector("[data-prev-shot]").addEventListener("click", () => {
        shotIndex[project] = (shotIndex[project] - 1 + projectShots[project].length) % projectShots[project].length;
        updateShot(project);
      });
      nav.querySelector("[data-next-shot]").addEventListener("click", () => {
        shotIndex[project] = (shotIndex[project] + 1) % projectShots[project].length;
        updateShot(project);
      });
    });

    // Keyboard support
    document.addEventListener("keydown", (e) => {
      if (!document.getElementById("view-digital-lab") || document.getElementById("view-digital-lab").classList.contains("hidden")) return;
      if (e.key === "ArrowLeft") goTo(current - 1);
      if (e.key === "ArrowRight") goTo(current + 1);
    });

    goTo(0);
  })();

  // Formspree handles submission — no mailto interceptor needed
  renderRoute();
</script>
